float roomTemp, humidity, bodyTemp;
int heartRate = 0;
int spo2 = 0;

unsigned long lastSend = 0;
unsigned long lastAlert = 0;

/******************** BUFFERS ********************/
#define BUFFER_SIZE 100
uint32_t irBuffer[BUFFER_SIZE];
uint32_t redBuffer[BUFFER_SIZE];

/******************** BODY TEMP AVG ********************/
float avgBodyTemp() {
  float sum = 0;
  for (int i = 0; i < 5; i++) {
    sensors.requestTemperatures();
    sum += sensors.getTempCByIndex(0);
    delay(200);
  }
  return sum / 5.0;
}

/******************** MAX30102 ********************/
void readMAX30102() {
  for (byte i = 0; i < BUFFER_SIZE; i++) {
    while (!particleSensor.available())
      particleSensor.check();

    redBuffer[i] = particleSensor.getRed();
    irBuffer[i] = particleSensor.getIR();
    particleSensor.nextSample();
  }

  int32_t spo2Val;
  int8_t spo2Valid;
  int32_t hrVal;
  int8_t hrValid;

  maxim_heart_rate_and_oxygen_saturation(
    irBuffer, BUFFER_SIZE, redBuffer,
    &spo2Val, &spo2Valid,
    &hrVal, &hrValid
  );

  if (spo2Valid) spo2 = spo2Val;
  if (hrValid) heartRate = hrVal;
}

/******************** GSM INIT ********************/
void initGSM() {
  gsm.println("AT");
  delay(1000);

  gsm.println("ATE0"); // Echo OFF
  delay(1000);

  gsm.println("AT+CMGF=1"); // SMS TEXT MODE
  delay(1000);

  gsm.println("AT+CSQ"); // Signal strength
  delay(1000);

  gsm.println("AT+CREG?"); // Network check
  delay(1000);
}

/******************** SEND SMS ********************/
void sendSMS(String msg) {
  gsm.println("AT+CMGF=1");
  delay(1000);

  gsm.println("AT+CMGS=\"+917569000477\"");
  delay(2000);

  gsm.print(msg);
  delay(500);

  gsm.write(26); // CTRL+Z
  delay(5000);
}

/******************** MAKE CALL ********************/
void makeCall() {
  gsm.println("ATD+917569000477;");
  delay(20000); // 20 sec call
  gsm.println("ATH");
}

/******************** LCD ********************/
void displayData() {

  lcd.clear();

  // ALERT SCREEN
  if (bodyTemp > 38.5) {
    lcd.setCursor(0, 0);
    lcd.print("!!! ALERT !!!");

    lcd.setCursor(0, 1);
    lcd.print("BODY TEMP HIGH");
    return;
  }

  // NORMAL DISPLAY
  lcd.setCursor(0, 0);
  lcd.print("HR:");
  lcd.print(heartRate);
  lcd.print(" SpO2:");
  lcd.print(spo2);

  lcd.setCursor(0, 1);
  lcd.print("BT:");
  lcd.print(bodyTemp, 1);
  lcd.print(" H:");
  lcd.print(humidity, 0);
}

/******************** THINGSPEAK ********************/
void sendToThingSpeak() {
  if (WiFi.status() == WL_CONNECTED) {

    HTTPClient http;

    String url = server;
    url += "?api_key=" + apiKey;
    url += "&field1=" + String(heartRate);
    url += "&field2=" + String(spo2);
    url += "&field3=" + String(bodyTemp);
    url += "&field4=" + String(roomTemp);
    url += "&field5=" + String(humidity);

    http.begin(url);
    http.GET();
    http.end();
  }
}

/******************** SETUP ********************/
void setup() {

  Serial.begin(115200);

  dht.begin();
  sensors.begin();

  lcd.init();
  lcd.backlight();

  gsm.begin(9600, SERIAL_8N1, GSM_RX, GSM_TX);
  initGSM();

  Wire.begin(21, 22);

  /******** MAX30102 ********/
  if (!particleSensor.begin(Wire, I2C_SPEED_FAST)) {
    Serial.println("MAX30102 not found");
    while (1);
  }
  particleSensor.setup();

  lcd.setCursor(0, 0);
  lcd.print("Health Monitor");
  lcd.setCursor(0, 1);
  lcd.print("Initializing...");
  delay(2000);

  /******** WIFI ********/
  WiFi.begin(ssid, password);
  lcd.clear();
  lcd.print("Connecting WiFi");

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    lcd.print(".");
  }

  lcd.clear();
  lcd.print("WiFi Connected");
  delay(1500);
}

/******************** LOOP ********************/
void loop() {

  /******** DHT ********/
  humidity = dht.readHumidity();
  roomTemp = dht.readTemperature();

  if (isnan(humidity) || isnan(roomTemp)) {
    humidity = 0;
    roomTemp = 0;
  }

  /******** BODY TEMP ********/
  bodyTemp = avgBodyTemp();

  /******** MAX30102 ********/
  readMAX30102();

  /******** DISPLAY ********/
  displayData();

  /******** GSM ALERT ********/
  if (bodyTemp > 38.5 && millis() - lastAlert > 60000) {

    String msg = "P13:ALERT!\nBT:" + String(bodyTemp) +
                 "\nHR:" + String(heartRate) +
                 "\nSpO2:" + String(spo2);

    sendSMS(msg);
    delay(3000);
    makeCall();

    lastAlert = millis();
  }

  /******** THINGSPEAK (5 sec) ********/
  if (millis() - lastSend > 3000) {
    sendToThingSpeak();
    lastSend = millis();
  }

  delay(2000);
}
